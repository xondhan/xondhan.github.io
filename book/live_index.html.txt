<head>

  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Live</title>
  <style>
    body { display: flex; flex-direction: column; align-items: center; justify-content: center; height: 100vh; margin: 0; }
    #notepad { width: 93%; height: 70vh; }
    button { margin-top: 10px; }
  </style>
</head>
<body>
  <textarea id="notepad" name="text"></textarea>
  <button onclick="saveText()">Save Now</button>

  <script>
    document.addEventListener('DOMContentLoaded', function() {
      const textarea = document.getElementById('notepad');

      fetch('get_existing_text.php')
        .then(response => response.text())
        .then(data => { textarea.value = data; })
        .catch(error => console.error('Error:', error));
    });

    function saveText() {
      const textarea = document.getElementById('notepad');
      const text = textarea.value;

      fetch('save.php', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: 'text=' + encodeURIComponent(text),
      })
      .then(() => { location.reload(); })
      .catch(error => console.error('Error:', error));
    }
  </script>
</body>
